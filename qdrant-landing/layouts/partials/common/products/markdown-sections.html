{{- /* For the Markdown output: wrap each section whose heading has a products or version attribute, heading included, in paired tags, and drop those attributes from the heading. Other heading attributes are kept. A section runs to the next heading of the same or a higher level, so subsections stay inside its tags. ## Quantization {products="clusters[1.20.0], edge[0.9.0]"} ... ## Next section becomes ## Quantization ... ## Next section Lines inside fenced code blocks are skipped, so a "# comment" in a shell snippet is not taken for a heading. Input: Markdown source (after .RenderShortcodes). Returns the rewritten Markdown. */ -}} {{- $out := string . -}} {{- $tagged := `^(#{1,6} [^\n]*?)[ \t]*\{([^}\n]*\b(?:products|version)="[^}\n]*)\}[ \t]*$` -}} {{- $attr := `\b(products|version)="([^"]*)"` -}} {{- if findRE (printf "(?m)%s" $tagged) $out 1 -}} {{- $lines := slice -}} {{- $open := slice -}} {{- $fence := "" -}} {{- range split $out "\n" -}} {{- $line := . -}} {{- $marker := "" -}} {{- with findRESubmatch "^[ \t]*(`{3,}|~{3,})" $line 1 }}{{ $marker = index (index . 0) 1 }}{{ end -}} {{- if $fence -}} {{- if and $marker (hasPrefix $marker $fence) }}{{ $fence = "" }}{{ end -}} {{- $lines = $lines | append $line -}} {{- else if $marker -}} {{- $fence = $marker -}} {{- $lines = $lines | append $line -}} {{- else -}} {{- with findRESubmatch `^(#{1,6})[ \t]` $line 1 -}} {{- $level := len (index (index . 0) 1) -}} {{- $keep := slice -}} {{- range $open }}{{ if lt . $level }}{{ $keep = $keep | append . }}{{ end }}{{ end -}} {{- range $open }}{{ if ge . $level }}{{ $lines = $lines | append "" "" }}{{ end }}{{ end -}} {{- $open = $keep -}} {{- with findRESubmatch $tagged $line 1 -}} {{- $m := index . 0 -}} {{- $attrs := dict -}} {{- range $lines = $lines | append "" }}{{ end -}} {{- $out = delimit $lines "\n" -}} {{- end -}} {{- return $out -}}
