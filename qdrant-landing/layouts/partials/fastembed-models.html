{{- /* type: "version", a class name from data/fastembed_models.json, or "others" for each class the page doesn't name. */ -}} {{- $catalog := hugo.Data.fastembed_models -}} {{- $type := .type | default "" -}} {{- $result := "" -}} {{- if eq $type "version" -}} {{- $result = (printf "These tables list the built-in models in [FastEmbed %s](https://github.com/qdrant/fastembed/releases/tag/v%s), grouped by the class that loads them. Model names, dimensions, and licenses come from that release's model registry. Your installed version determines which models you can use." $catalog.version $catalog.version) -}} {{- else if eq $type "others" -}} {{- $named := slice -}} {{- range findRESubmatch `fastembed-models\s+type\s*=\s*["']?(\w+)` .page.RawContent -}} {{- $named = $named | append (index . 1) -}} {{- end -}} {{- $sections := slice -}} {{- range $catalog.classes -}} {{- if not (in $named .name) -}} {{- $table := partial "fastembed-models.html" (dict "type" .name "page" $.page) -}} {{- $sections = $sections | append (printf "## %s Models\n\nLoad these models with the `%s` class.\n\n%s" .name .name $table) -}} {{- end -}} {{- end -}} {{- $result = delimit $sections "\n\n" -}} {{- else -}} {{- $models := slice -}} {{- range where $catalog.classes "name" $type -}}{{- $models = .models -}}{{- end -}} {{- if not $models -}} {{- errorf "fastembed-models: no %q models in data/fastembed_models.json" $type -}} {{- end -}} {{- $hasDimensions := isset (index $models 0) "dimensions" -}} {{- $rows := slice (printf "| Model |%s License |" (cond $hasDimensions " Dimensions |" "")) (printf "|---|%s---|" (cond $hasDimensions "---|" "")) -}} {{- range $models -}} {{- if not (and .model .license) -}} {{- errorf "fastembed-models: each %s model needs a model name and license" $type -}} {{- end -}} {{- $dimensions := "" -}} {{- if $hasDimensions -}} {{- if not .dimensions -}}{{- errorf "fastembed-models: %s needs dimensions" .model -}}{{- end -}} {{- $dimensions = printf " %v |" .dimensions -}} {{- end -}} {{- $rows = $rows | append (printf "| `%s` |%s %s |" .model $dimensions .license) -}} {{- end -}} {{- $result = delimit $rows "\n" -}} {{- end -}} {{- return $result -}}
