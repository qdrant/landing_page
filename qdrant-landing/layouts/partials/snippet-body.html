{{- /* Assemble the all-languages markdown body for a (dir, block) pair. Params: dir (fs path), block (name or ""), order (slice of langs). Mirrors layouts/shortcodes/code-snippet.markdown.md. Returns markdown. */ -}} {{- $dir := .dir -}} {{- $block := .block -}} {{- $order := .order -}} {{- $files := dict -}} {{- range slice $dir (printf "%s/generated/%s" $dir $block) -}} {{- $d := . -}} {{- if fileExists $d -}} {{- range (readDir $d) -}} {{- if and (strings.HasSuffix .Name ".md") (not (strings.HasPrefix .Name "_")) (ne .Name "index.md") -}} {{- $lang := strings.TrimSuffix ".md" .Name -}} {{- $files = merge $files (dict $lang (readFile (printf "%s/%s" $d .Name))) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $parts := slice -}} {{- range $order -}} {{- with index $files . -}}{{- $parts = $parts | append . -}}{{- end -}} {{- end -}} {{- range $lang, $content := $files -}} {{- if not (in $order $lang) -}}{{- $parts = $parts | append $content -}}{{- end -}} {{- end -}} {{- return (delimit $parts "\n") -}}
