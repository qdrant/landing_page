{{- /* Publish a page's Markdown output a second time as /path.md, the guessable form a client can build from a URL without fetching the HTML first. Additions, never moves: the index.md URL stays where it is. Recorded in hugo.Store so index.netlifyheaders writes a Link header for exactly the aliases that were published. Skipped for the site root, which has no /path.md form, and where static/ already holds a file at that path. A template rather than a Netlify redirect because a splat has to be terminal, so /articles/*.md is not a pattern Netlify accepts. */ -}} {{- with .page.OutputFormats.Get "markdown" -}} {{- $rel := .RelPermalink -}} {{- if and (strings.HasSuffix $rel "/index.md") (ne $rel "/index.md") -}} {{- $alias := printf "%s.md" (strings.TrimSuffix "/index.md" $rel) -}} {{- if not (fileExists (printf "static%s" $alias)) -}} {{- $_ := (resources.FromString (strings.TrimPrefix "/" $alias) $.content).Publish -}} {{- hugo.Store.SetInMap "markdownAliases" (strings.TrimSuffix "index.md" $rel) $alias -}} {{- end -}} {{- end -}} {{- end -}}
