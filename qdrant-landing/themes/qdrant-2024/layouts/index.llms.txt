{{- define "_partials/llms-tree" -}}
{{- $p := .page -}}
{{- $listed := and $p (not $p.Params.sitemapExclude) (not $p.Sitemap.Disable) -}}
{{- $rendered := $listed -}}
{{- with and $listed $p.Params.build -}}
  {{- if in (slice "never" "link") (.render | default "always") -}}{{- $rendered = false -}}{{- end -}}
{{- end -}}
{{- $indent := .indent -}}
{{- if $rendered -}}
{{- if eq $p.Params.type "external-link" -}}
{{ .indent }}- [{{ $p.Title }}]({{ $p.Params.external_url }}){{ with partial "llms-summary" $p }}: {{ . }}{{ end }}
{{ else if and (ne $p.Title "") (ne $p.Permalink "") -}}
{{ .indent }}{{ partial "llms-entry" $p }}
{{- end -}}
{{- $indent = printf "%s  " .indent -}}
{{- end -}}
{{- if and $listed (or (not .flat) (not $rendered)) -}}
{{- $flat := .flat -}}
{{- range $p.Pages.ByWeight }}{{ partial "llms-tree" (dict "page" . "indent" $indent "flat" $flat) }}{{ end -}}
{{- end -}}
{{- end -}}

{{- /* One link with its one-line summary. */ -}}
{{- define "_partials/llms-entry" -}}

{{ end -}}

{{- /* short_description, else the first sentence of the description, else of
       the content. The last case warns once per page. */ -}}
{{- define "_partials/llms-summary" -}}
{{- $clean := `\s+` -}}
{{- $d := .Params.short_description | default "" | plainify | htmlUnescape | replaceRE $clean " " | strings.TrimSpace -}}
{{- if not $d -}}
  {{- $d = .Description | plainify | htmlUnescape | replaceRE $clean " " | strings.TrimSpace -}}
  {{- if not $d -}}
    {{- /* Headings are dropped first, or the page title leads the sentence. */ -}}
    {{- $d = .Content | replaceRE `(?s)<h[1-6][^>]*>.*?</h[1-6]>` "" | plainify | htmlUnescape | replaceRE $clean " " | strings.TrimSpace -}}
    {{- $key := printf "llmsNoSummary%s" .RelPermalink -}}
    {{- if not (hugo.Store.Get $key) -}}
      {{- warnf "llms.txt: %s has no short_description or description; its summary comes from the page content" .File.Path -}}
      {{- hugo.Store.Set $key true -}}
    {{- end -}}
  {{- end -}}
  {{- $d = index (split $d ". ") 0 | strings.TrimSuffix "." -}}
{{- end -}}
{{- $d -}}
{{- end -}}
