{{- /* chart — embed a pre-generated chart SVG by id. The SVG is built by scripts/viz/generate-charts.mjs and committed, not rendered at request time. Charts need Observable Plot and a DOM, which is a Node toolchain Hugo does not have; committing the output also means a chart cannot silently change when a dependency updates, and every diff to a number shows up in review. Every lookup below fails the BUILD rather than degrading: a typo'd id, a missing manifest entry or a stale SVG would otherwise ship as a blank gap in a published post, which nobody notices until a reader does. The error names the fix, including the exact command to regenerate. Usage: a "chart" shortcode taking id= and caption=, where id is the path to a chart's three files: assets/viz/.{json,csv,svg}. */ -}} {{- $id := .Get "id" -}} {{- if not $id }}{{ errorf "chart in %s: 'id' is required." .Position }}{{ end -}} {{- $caption := .Get "caption" -}} {{- if not $caption }}{{ errorf "chart in %s: 'caption' is required." .Position }}{{ end -}} {{- $specRes := resources.Get (printf "viz/%s.json" $id) -}} {{- if not $specRes }}{{ errorf "chart in %s: assets/viz/%s.json missing." .Position $id }}{{ end -}} {{- $spec := $specRes | transform.Unmarshal -}} {{- $res := resources.Get (printf "viz/%s.svg" $id) -}} {{- if not $res }}{{ errorf "chart in %s: assets/viz/%s.svg missing. Run: node scripts/viz/generate-charts.mjs" .Position $id }}{{ end -}} {{- /* Space the generator draws into below the plot; the viewBox must include it or the legend is clipped. Sizes live in data/viz.json. */ -}} {{- $chart := site.Data.viz.chart -}} {{- $room := 0 -}} {{- if eq $spec.kind "grouped-columns" -}} {{- $room = $chart.legendRoom -}} {{- else if eq $spec.kind "lines-facet" -}} {{- $room = cond (gt (len $spec.series) 1) $chart.legendRoom $chart.facetRoom -}} {{- end -}} {{- $views := slice -}} {{- range $spec.views -}}{{- $views = $views | append .label -}}{{- end -}} {{- partial "viz-figure.html" (dict "svg" $res.Content "viewBox" (printf "0 0 %d %d" (int $chart.width) (add (int $spec.height) (int $room))) "caption" $caption "id" (replace $id "/" "-") "class" "viz-figure--chart" "views" $views) -}}
