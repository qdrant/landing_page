{{- /*
  A page as a link, then every page under it, nested, unless `flat`. With
  `desc`, each link carries the first sentence of the page's description.
  A section that is never rendered itself (an `is_empty` group such as the
  FAQ) is not listed, but its pages are, even when `flat`.
*/ -}}
{{- define "_partials/llms-tree" -}}
{{- $p := .page -}}
{{- $listed := and $p (not $p.Params.sitemapExclude) (not $p.Sitemap.Disable) -}}
{{- $rendered := $listed -}}
{{- with and $listed $p.Params.build -}}
  {{- if in (slice "never" "link") (.render | default "always") -}}{{- $rendered = false -}}{{- end -}}
{{- end -}}
{{- $indent := .indent -}}
{{- if $rendered -}}
{{- if eq $p.Params.type "external-link" -}}
{{ .indent }}- [{{ $p.Title }}]({{ $p.Params.external_url }})
{{ else if and (ne $p.Title "") (ne $p.Permalink "") -}}
{{ .indent }}{{ partial "llms-entry" (dict "page" $p "desc" .desc) }}
{{- end -}}
{{- $indent = printf "%s  " .indent -}}
{{- end -}}
{{- if and $listed (or (not .flat) (not $rendered)) -}}
{{- $desc := .desc -}}{{- $flat := .flat -}}
{{- range $p.Pages.ByWeight }}{{ partial "llms-tree" (dict "page" . "indent" $indent "desc" $desc "flat" $flat) }}{{ end -}}
{{- end -}}
{{- end -}}

{{- /* One link. Given a page, or a dict with `page` and `desc`: the description
       always for a bare page, its first sentence only for `desc`. */ -}}
{{- define "_partials/llms-entry" -}}
{{- $p := . -}}{{- $short := false -}}{{- $withDesc := true -}}
{{- if reflect.IsMap . }}{{ $p = .page }}{{ $short = true }}{{ $withDesc = .desc }}{{ end -}}
{{- $d := $p.Description | plainify | replaceRE "\\s+" " " | strings.TrimSpace -}}
{{- if $short }}{{ $d = index (split $d ". ") 0 | strings.TrimSuffix "." }}{{ end -}}
- [{{ $p.Title }}]({{ strings.TrimSuffix "/" $p.Permalink }}.md){{ if and $withDesc $d }}: {{ $d }}{{ end }}
{{ end -}}

{{- /* Publish a part's full index at /llms/<key>.txt and link it. */ -}}
{{- define "_partials/llms-index-link" -}}
{{- $name := .part.name -}}
{{- $count := len (findRE `(?m)^\s*- \[` .body) -}}
{{- $file := printf "%s\n\n> Every page in the %s part of qdrant.tech, nested in sidebar order, each with a one-line summary. Links point to the Markdown version of each page. For setup, core usage, and rules, start at %s.\n%s" (printf "# Qdrant %s: every page" $name) $name (absURL "llms.txt") .body -}}
{{- $path := printf "llms/%s.txt" .part.key -}}
{{- $_ := (resources.FromString $path $file).Publish -}}
- [{{ .part.label | default (printf "Every %s page" $name) }}]({{ absURL $path }}): all {{ $count }} pages, each with a one-line summary
{{ end -}}
