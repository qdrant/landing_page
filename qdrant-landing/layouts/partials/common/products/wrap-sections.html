Content without tagged headings is returned unchanged. */ -}} {{- $out := string . -}} {{- if strings.Contains $out " data-products=" -}} {{- $parts := split (replaceRE `])` "\x1e" $out }}{{ end -}} {{- end -}} {{- $stack = $keep -}} {{- $tag := findRESubmatch `^]*\sdata-products="([^"]*)"` $part 1 -}} {{- if $tag -}} {{- $r := partial "common/products/resolve.html" (dict "products" (split (index (index $tag 0) 1) " ") "where" "wrap-sections") -}} {{- $heading := index (findRE `(?s)^]*>.*?` $part 1) 0 -}} {{- $rest := strings.TrimPrefix $heading $part -}} {{- $out = printf "%s
%s%s
%s" $out (delimit $r.keys " ") $heading (partial "common/products/note.html" $r) $rest -}} {{- $stack = $stack | append $level -}} {{- else -}} {{- $out = printf "%s%s" $out $part -}} {{- end -}} {{- end -}} {{- range $stack }}{{ $out = printf "%s
" $out }}{{ end -}} {{- end -}} {{- return $out -}}
